<!-- glass card -->
<div class="glass-card">
  <h3>Frosted Panel</h3>
  <p>Hover to see the glow</p>
</div>
.glass-card {
  background: rgba(255,255,255,.06);
  border: 1px solid rgba(255,255,255,.15);
  backdrop-filter: blur(20px);
  border-radius: 16px;
  transition: transform .3s ease;
}
.glass-card:hover {
  transform: translateY(-6px);
  box-shadow: 0 20px 40px -10px rgba(91,140,255,.4);
}
// optional tilt on mouse move
card.addEventListener('mousemove', (e) => {
  // compute rotation from cursor position
  const rx = (e.offsetY / card.clientHeight - 0.5) * -12;
  card.style.transform = 'rotateX(' + rx + 'deg)';
});
